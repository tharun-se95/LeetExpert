# Lamp Logo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace both shipping brand marks (the header/hero `</>` badge and the brackets-and-check icon set) with the lamp — one soft-glowing grass-green dot — on every surface.

**Architecture:** Lamp colours are theme tokens in `globals.css`. A small client component (`Lamp.tsx`) draws the mark as inline SVG painted only from those tokens. A pre-paint script marks the first page of each tab so a CSS-only 900ms bloom runs from the first frame. Static surfaces (SVG/PNG/ICO favicons, home-screen tiles, the OG card) use hex that mirrors the tokens. A new test suite enforces contrast, token scope, token-to-asset sync, and that the icon pipeline actually ran.

**Tech Stack:** Next.js 16 App Router, React 19.2, Tailwind v4 (tokens in `globals.css`), Vitest 4 (node env, `renderToStaticMarkup`), sharp 0.34.5, next/og (Satori).

**Spec:** `docs/superpowers/specs/2026-09-13-lamp-logo-design.md` — read the "Amendments during planning" section; this plan implements the amended spec.

```ts
/**
 * WCAG 2.x relative luminance and contrast ratio. Shared by the token
 * suites so a correction to the formula lands everywhere at once — two
 * copies of a contrast function are two chances to certify a failing ink.
 */
export function relativeLuminance(hex: string): number {
  const h = hex.replace("#", "");
  const full =
    h.length === 3
      ? h
          .split("")
          .map((c) => c + c)
          .join("")
      : h;
  const channels = [0, 2, 4].map((i) => {
    const c = parseInt(full.slice(i, i + 2), 16) / 255;
    return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * channels[0]! + 0.7152 * channels[1]! + 0.0722 * channels[2]!;
}

export function contrastRatio(a: string, b: string): number {
  const L1 = relativeLuminance(a);
  const L2 = relativeLuminance(b);
  const [hi, lo] = L1 > L2 ? [L1, L2] : [L2, L1];
  return (hi + 0.05) / (lo + 0.05);
}
```

```tsx
import { describe, it, expect } from "vitest";
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative } from "node:path";
import { contrastRatio } from "./helpers/contrast";

/**
 * The brand lamp: one soft-glowing green dot is the whole logo, and the only
 * non-steel ink allowed in chrome.
 *
 * Spec: docs/superpowers/specs/2026-09-13-lamp-logo-design.md
 */

const WEB = join(__dirname, "..");
const SRC = join(WEB, "src");
const css = readFileSync(join(SRC, "app", "globals.css"), "utf8");
// Same slicing as design-tokens.test.ts: the first :root up to the first
// .dark is the light palette, the first .dark onward is dark. globals.css
// has later :root/.dark blocks too — the lamp tokens must live in the FIRST
// of each or every reader of this convention sees the wrong theme.
const root = css.slice(css.indexOf(":root"), css.indexOf(".dark"));
const dark = css.slice(css.indexOf(".dark"));

function hexVar(block: string, name: string): string {
  const m = block.match(new RegExp(`--${name}:\\s*(#[0-9a-fA-F]{6})\\b`));
  if (!m?.[1]) throw new Error(`--${name} has no 6-digit hex in this block`);
  return m[1].toLowerCase();
}

function walk(dir: string): string[] {
  const out: string[] = [];
  for (const entry of readdirSync(dir)) {
    if (entry === "node_modules" || entry.startsWith(".")) continue;
    const full = join(dir, entry);
    if (statSync(full).isDirectory()) out.push(...walk(full));
    else if (/\.(tsx?|css)$/.test(entry)) out.push(full);
  }
  return out;
}

describe("brand lamp tokens", () => {
  it("the lamp clears the 3:1 non-text floor on every surface, both themes", () => {
    // A logo is a meaningful graphic (WCAG 1.4.11), so the floor is 3:1, not
    // the 4.5:1 text floor — and it is checked against every surface token,
    // not just the header it happens to sit on today.
    const themes = [
      {
        theme: "light",
        lamp: hexVar(root, "press-lamp"),
        surfaces: {
          elevated: hexVar(root, "elevated"),
          paper: hexVar(root, "press-paper"),
          code: hexVar(root, "code"),
          sunk: hexVar(root, "press-paper-sunk"),
        },
      },
      {
        theme: "dark",
        lamp: hexVar(dark, "press-lamp"),
        surfaces: {
          elevated: hexVar(dark, "elevated"),
          paper: hexVar(dark, "press-paper"),
          code: hexVar(dark, "code"),
          surface: hexVar(dark, "surface"),
        },
      },
    ];
    for (const { theme, lamp, surfaces } of themes) {
      for (const [name, surface] of Object.entries(surfaces)) {
        const ratio = contrastRatio(lamp, surface);
        expect(
          ratio,
          `${theme} lamp ${lamp} on ${name} ${surface} is ${ratio.toFixed(2)}:1, needs >= 3:1`,
        ).toBeGreaterThanOrEqual(3);
      }
    }
  });

  it("lamp colour is confined to the mark", () => {
    // The lamp is a deliberate exception to the monochrome base. Without a
    // guard, the next contributor who needs "a green" reaches for it and
    // brand colour leaks into buttons, focus rings and status.
    const LAMP_TOKEN = /--(?:press-)?lamp(?:-[a-z]+)?\b/;
    const LAMP_UTILITY =
      /\b(?:bg|text|fill|stroke|border|ring|outline|decoration|from|via|to|shadow|accent|caret)-lamp(?:-[a-z]+)?\b/;

    const leaks = walk(SRC)
      .map((path) => ({
        rel: relative(SRC, path).replace(/\\/g, "/"),
        body: readFileSync(path, "utf8"),
      }))
      .filter(
        (f) =>
          !OWNERS.has(f.rel) &&
          (LAMP_TOKEN.test(f.body) || LAMP_UTILITY.test(f.body)),
      )
      .map((f) => f.rel);
```

```tsx
"use client";

import { useId, type AnimationEvent } from "react";

interface LampProps {
  /** Rendered width and height in px; the mark is drawn on a 64 grid. */
  size: number;
  /**
   * Power-on bloom: plays on the first paint of a tab (see lampBoot.ts) and
   * on hover/focus of an ancestor `.lamp-link`. Only the header lamp blooms.
   */
  bloom?: boolean;
  className?: string;
}

/**
 * The leetExpert mark: a soft-glowing dot. Painted only from the --lamp
 * tokens so both themes come from globals.css.
 * Spec: docs/superpowers/specs/2026-09-13-lamp-logo-design.md
 */
export function Lamp({ size, bloom = false, className }: LampProps) {
  // useId's output format has changed across React versions and can include
  // characters that break a url(#…) reference, so keep only id-safe ones.
  const uid = useId().replace(/[^a-zA-Z0-9_-]/g, "");
  const haloId = `lamp-halo-${uid}`;
  const coreId = `lamp-core-${uid}`;

  // Clear the first-paint flag once the bloom finishes. Left in place, a
  // hover that ends would drop the replay animation, fall back to the boot
  // selector, and play the bloom a second time.
  function handleAnimationEnd(event: AnimationEvent<SVGSVGElement>) {
    if (event.animationName.startsWith("lamp-halo")) {
      delete document.documentElement.dataset.lampBoot;
    }
  }

  return (
    <svg
      width={size}
      height={size}
      viewBox="0 0 64 64"
      aria-hidden
      focusable="false"
      className={["lamp", bloom ? "lamp--bloom" : null, className]
        .filter(Boolean)
        .join(" ")}
      onAnimationEnd={bloom ? handleAnimationEnd : undefined}
    >
      <defs>
```
